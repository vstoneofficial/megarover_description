#define _GNU_SOURCE
#include <arpa/inet.h>
#include <dlfcn.h>
#include <errno.h>
#include <stdio.h>
#include <sys/socket.h>
#include <unistd.h>

int bind(int fd, const struct sockaddr *address, socklen_t length)
{
  static int (*original)(int, const struct sockaddr *, socklen_t);
  if (!original) original = dlsym(RTLD_NEXT, "bind");
  int result = original(fd, address, length);
  int error = errno;
  if (result < 0 && error != EADDRINUSE) {
    unsigned port = address && address->sa_family == AF_INET ?
      ntohs(((const struct sockaddr_in *)address)->sin_port) : 0;
    dprintf(STDERR_FILENO, "BIND_DIAGNOSTIC pid=%d fd=%d port=%u errno=%d\n",
      getpid(), fd, port, error);
  }
  errno = error;
  return result;
}
